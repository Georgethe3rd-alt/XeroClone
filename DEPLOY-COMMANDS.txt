# Dashboard Deployment Commands
# Run these from the Docker host (the machine running both containers)

# Step 1: Copy deployment package to Dashboard container
docker cp /tmp/dashboard-complete.tar.gz dashboard:/tmp/

# Step 2: Extract files and install dependencies
docker exec dashboard bash -c "cd /app && tar xzf /tmp/dashboard-complete.tar.gz && npm install --omit=dev"

# Step 3: Set environment variables (if needed)
docker exec dashboard bash -c "export WEBHOOK_URL='http://172.18.0.2:3005' && export DASH_PASSWORD='Wayne2026#'"

# Step 4: Restart dashboard
docker restart dashboard

# Step 5: Test (wait 5 seconds after restart)
sleep 5
curl -X POST http://172.18.0.1/api/auth -H "Content-Type: application/json" -d '{"password":"Wayne2026#"}'

# If you see a token in the response, deployment succeeded!

---

# Alternative: All in one command
docker cp /tmp/dashboard-complete.tar.gz dashboard:/tmp/ && \
docker exec dashboard bash -c "cd /app && tar xzf /tmp/dashboard-complete.tar.gz && npm install --omit=dev" && \
docker restart dashboard && \
sleep 5 && \
curl -X POST http://172.18.0.1/api/auth -H "Content-Type: application/json" -d '{"password":"Wayne2026#"}'

---
